# Commande: @{ name = 'a' ; weight = 7 } | Select-Object -Property @(@{ Label = 'Name' ; Expression = { $_.name } }, @{ Label = 'Weight' ; Expression = { $_.weight } })
Name Weight
---- ------
a         7
